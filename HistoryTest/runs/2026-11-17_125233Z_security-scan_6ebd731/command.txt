import json,urllib.request
q=open('/tmp/claude-0/osvin/osv_q.json','rb').read()

n=0
for a,b in zip(json.loads(q)['queries'],r):
  v=[x['id'] for x in b.get('vulns',[])]; n+=len(v); print(a['package']['name'],a['version'],v or 'none')
print('total advisories:',n); raise SystemExit(1 if n else 0)
